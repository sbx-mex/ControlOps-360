let
    Origen =
        Sql.Database(
            "localhost\sqlexpress",
            "BI_ALSEA",
            [
                Query = "DECLARE @Desde DATE = DATEADD(DAY, -20, CONVERT(DATE, GETDATE()));

WITH Producto AS
(
    SELECT
        P.IDProducto,
        P.ClaveProducto,
        P.IDTamanio,
        P.IDFamilia,
        P.DescripcionFam,
        P.Descripcion,
        ROW_NUMBER() OVER
        (
            PARTITION BY P.IDProducto
            ORDER BY P.FechaNegocio DESC, P.ClaveProducto DESC
        ) AS rn
    FROM dbo.Vta_Productos AS P
)
SELECT
    D.IDTienda,
    D.FechaHora,
    D.Ticket,
    D.SecTrans,
    D.SecDtl,
    D.Id,
    D.IDProducto,
    D.Cantidad,
    D.CantidadAjustada,
    D.Total,
    D.NivelPrecio,
    D.ModoOrden,
    D.ModoOrdenDesc,
    D.IDEmpleado,
    D.IdTerminal,
    P.ClaveProducto AS CodigoDIA,
    P.IDTamanio,
    P.IDFamilia,
    P.DescripcionFam,
    P.Descripcion
FROM dbo.VTA_DetalleTicket AS D
LEFT JOIN Producto AS P
    ON P.IDProducto = D.IDProducto
   AND P.rn = 1
WHERE D.FechaHora >= @Desde
  AND D.IDProducto IS NOT NULL
OPTION (RECOMPILE);"
            ]
        ),
    #"Tipos definidos" =
        Table.TransformColumnTypes(
            Origen,
            {
                {"IDTienda", Int64.Type},
                {"FechaHora", type datetime},
                {"Ticket", Int64.Type},
                {"SecTrans", Int64.Type},
                {"SecDtl", Int64.Type},
                {"Id", Int64.Type},
                {"IDProducto", Int64.Type},
                {"Cantidad", type number},
                {"CantidadAjustada", type number},
                {"Total", Currency.Type},
                {"NivelPrecio", Int64.Type},
                {"ModoOrden", Int64.Type},
                {"ModoOrdenDesc", type text},
                {"IDEmpleado", Int64.Type},
                {"IdTerminal", Int64.Type},
                {"CodigoDIA", Int64.Type},
                {"IDTamanio", Int64.Type},
                {"IDFamilia", Int64.Type},
                {"DescripcionFam", type text},
                {"Descripcion", type text}
            }
        )
in
    #"Tipos definidos"
